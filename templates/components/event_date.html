{#- Renders an event's `extra.date` as prose, e.g. "Wednesday, October 14, 2026". The date must be a full "YYYY-MM-DD"; anything else fails the build. -#} {% component event_date(date: string) %} {%- set d = date | str -%} {%- if d is matching(pat="^[0-9]{4}-[0-9]{2}-[0-9]{2}$") -%} {{- d | date(format="%A, %B %-d, %Y") -}} {%- else -%} {{ throw(message="Event date must be YYYY-MM-DD; got \"" ~ d ~ "\"") }} {%- endif -%} {% endcomponent event_date %}
